Footer 层分支 if (template === "signature") { const isDark = await isFooterAreaDark(background, layout); footerLayer = await renderSignature(layout, isDark); // SVG 内存缓存 } else { footerLayer = await renderTypography(layout, photo, theme); } // 4. 并行渲照片 + 叠加层 const [overlay, photoCard] = await Promise.all([renderOverlay(...), renderPhoto(...)]); // 5. 合成四层 const buffer = await renderComposite(background, overlay, photoCard, footerLayer, layout, theme); return { buffer, layout }; }

2. Blob URL 泄漏

3. IntersectionObserver 膨胀

// ❌ 每个卡片独立 create → 100 张 = 100 个 Observer // ✅ 共享 hook，firedRef 防重复 export function useInViewOnce() { const ref = useCallback((el) => { if (!el || firedRef.current) return; const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) { firedRef.current = true; setInView(true); obs.disconnect(); } }, ...); obs.observe(el); }, []); return { ref, inView }; }
