Footer 层：经典 EXIF 或 签名 SVG if (template === "signature") { const isDark = await isFooterAreaDark(background, layout); footerLayer = await renderSignature(layout, isDark); } else { footerLayer = await renderTypography(layout, photo, theme); } // 4. 并行渲照片 + 叠加层 const [overlay, photoCard] = await Promise.all([ renderOverlay(layout, theme), renderPhoto(imageBuffer, layout, theme), // rotate+resize+clip+shadow+1px edge ]); // 5. 合成：bg → overlay → photoCard → footer const buffer = await renderComposite(background, overlay, photoCard, footerLayer, layout, theme); return { buffer, layout }; }

📚 技术经验与教训

1. SVG → librsvg 版本敏感

// ❌ VPS 失败（本地正常）： NIKON // ✅ 修复：所有属性移到父 ， 只保留 weight/size NIKON // 🐛 字体引号陷阱：fontFamily 里的 "SF Pro Display" 在被放入 // SVG 属性 font-family="..." 时 → 属性提前断裂 // 修复：去掉引号 → SF Pro Display

2. Blob URL 泄漏

// ❌ 切模板时旧 URL.revokeObjectURL 被遗漏 setShareUrl(URL.createObjectURL(blob)); // → 每次切换构建一个永不释放的 blob // ✅ 单 blobRef + seq 竞态 revoke(blobRef.current); blobRef.current = URL.createObjectURL(blob); // cleanup unmount → revoke(blobRef.current)

3. IntersectionObserver 膨胀

// ❌ 每张照片创建独立 observer → 100 张照片 = 100 个 Observer useEffect(() => { const obs = new IntersectionObserver(...); obs.observe(el); return () => obs.disconnect(); }, []); // ✅ 抽取共享 hook，firedRef 防重复 export function useInViewOnce() { const ref = useCallback((el) => { if (!el || firedRef.current) return; const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) { firedRef.current = true; setInView(true); obs.disconnect(); } }, ...); obs.observe(el); }, []); return { ref, inView }; }

4. 分享图生成耗时分析

总耗时 ~5.5s (R2 原图下载 ~3-4s + 毛玻璃 blur ~1s + 照片缩放 ~0.5s + 合成 ~0.2s) 瓶颈：原图在 R2 的 6000×4000 WebP（~3-5MB）。优化方向： - 上传时额外生成一个 2400px 的“分享用”副本 - 或在 share route 中用 thumbnailUrl 替代 photo.url

5. 上传取消 = XHR + AbortController
